🗄️ DATABASE SCHEMA

Database Name: ncc_chatbot

Table 1: conversations
  id (INT) - Primary Key
  session_id (VARCHAR, UNIQUE)
  user_name (VARCHAR)
  email (VARCHAR)
  created_at (TIMESTAMP)
  updated_at (TIMESTAMP)

Table 2: messages
  id (INT) - Primary Key
  conversation_id (INT) - Foreign Key
  sender (VARCHAR) - 'user' or 'bot'
  message (TEXT)
  created_at (TIMESTAMP)

Table 3: faq
  id (INT) - Primary Key
  category (VARCHAR)
  question (TEXT)
  answer (LONGTEXT)
  keywords (VARCHAR)
  created_at (TIMESTAMP)
  updated_at (TIMESTAMP)
